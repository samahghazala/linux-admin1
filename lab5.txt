Task 1: Group and User Creation 
 1. Create a new group named sysadmins and manually assign it the GID 35000.
root@localhost:~# groupadd -g 35000 sysadmins
root@localhost:~# tail -n3 /etc/group
devops_intern:x:4000:
sysadmin1:x:2001:
sysadmins:x:35000:

 2. Create two users named sysadmin1 and sysadmin2. Ensure that sysadmins is their supplementary group.
root@localhost:~# for user in sysadmin1 sysadmin2; do  useradd -G sysadmins "$user"; done
root@localhost:~# tail -n3 /etc/passwd
samah:x:1000:1000:samah:/home/samah:/bin/bash
sysadmin1:x:1001:1001::/home/sysadmin1:/bin/bash
sysadmin2:x:1002:1002::/home/sysadmin2:/bin/bash
root@localhost:~# id sysadmin1
uid=1001(sysadmin1) gid=1001(sysadmin1) groups=1001(sysadmin1),35000(sysadmins)
root@localhost:~# id sysadmin2
uid=1002(sysadmin2) gid=1002(sysadmin2) groups=1002(sysadmin2),35000(sysadmins)
root@localhost:~# 

 
 3. Set the initial password for both users to redhat.

passwd: password updated successfully
root@localhost:~# passwd sysadmin2

 Task 3: Shared Directory and Permissions 
 1. Create a directory at the path /opt/sysadmin_tools.
root@localhost:~# ls -l /opt
total 0
drwxr-xr-x. 2 root root 6 Sep 25 01:48 sysadmin_tools

 2. Change the group ownership of this directory so it is owned by the sysadmins group.
root@localhost:~# chgrp sysadmins /opt/sysadmin_tools/
root@localhost:~# ls -l /opt
total 0
drwxr-xr-x. 2 root sysadmins 6 Sep 25 01:48 sysadmin_tools
 
 3. Using the Symbolic Method, grant the group members write access to this directory.
root@localhost:~# chmod g+w /opt/sysadmin_tools/
root@localhost:~# ls -l /opt
total 0
drwxrwxr-x. 2 root sysadmins 6 Sep 25 01:48 sysadmin_tools
root@localhost:~# 


 4. Using the Octal Method, modify the permissions so that:
 ○ The Owner has full access (Read, Write, Execute). 
 ○ The Group has full access (Read, Write, Execute).
 ○ Others have no permissions (cannot read, write, or enter the directory).
 
root@localhost:~# chmod 770 /opt/sysadmin_tools/
root@localhost:~# ls -l /opt
total 0
drwxrwx---. 2 root sysadmins 6 Sep 25 01:48 sysadmin_tools
root@localhost:~#
